// ============================================= // ➕ SAVE PASSWORD (ফিক্সড) // ============================================= function savePassword() { const site = document.getElementById('siteName').value.trim(); const username = document.getElementById('siteUsername').value.trim(); const email = document.getElementById('siteEmail').value.trim(); const password = document.getElementById('sitePassword').value.trim(); if (!site || !password) { showToast('⚠️ সাইটের নাম এবং পাসওয়ার্ড দিন', 'error'); return; } const newEntry = { site: site, username: username, email: email, password: password, confirmed: false, createdAt: firebase.database.ServerValue.TIMESTAMP }; db.ref('users/' + currentUser.uid + '/passwords').once('value') .then(snapshot => { const passwords = snapshot.val() || []; passwords.push(newEntry); return db.ref('users/' + currentUser.uid + '/passwords').set(passwords); }) .then(() => { showToast('✅ তথ্য সেভ হয়েছে!', 'success'); // ===== ফিল্ড ক্লিয়ার ===== document.getElementById('siteSelect').value = ''; document.getElementById('siteName').value = ''; document.getElementById('siteUsername').value = ''; document.getElementById('siteEmail').value = ''; document.getElementById('sitePassword').value = ''; // ===== ফর্ম হাইড ===== document.getElementById('addPasswordForm').classList.add('hidden'); // ===== লিস্ট রিফ্রেশ ===== loadPasswords(); }) .catch(err => { showToast('❌ সেভ করতে সমস্যা: ' + err.message, 'error'); }); }
